{%- comment -%} One ordering and numbering implementation for the bibliography and citations. Pass category to render full entries, or ids to render numbered reference links. Use built-in Jekyll/Liquid filters for GitHub Pages compatibility. Group sort levels explicitly; chained sorts need not be stable. {%- endcomment -%} {%- assign reference_separator = "" -%} {%- if include.ids -%} {%- assign categories = "preprint,article" | split: "," -%} {%- else -%} {%- assign categories = include.category | split: "," -%} {%- endif -%} {%- for category in categories -%} {%- assign publications = site.data.publications | where: "category", category -%} {%- assign publication_number = publications.size -%} {%- if category == "preprint" -%} {%- assign articles = site.data.publications | where: "category", "article" -%} {%- assign publication_number = publication_number | plus: articles.size -%} {%- endif -%} {%- assign years = publications | group_by: "year" | sort: "name" | reverse -%} {%- for year in years -%} {%- assign author_groups = year.items | group_by_exp: "paper", "paper.authors | join: ' / ' | downcase | replace: 'ä', 'a' | replace: 'ö', 'o' | replace: 'ü', 'u' | replace: 'ć', 'c'" | sort: "name" -%} {%- for author_group in author_groups -%} {%- assign papers = author_group.items | sort_natural: "title" -%} {%- for publication in papers -%} {%- if include.ids -%} {%- assign reference_id = publication.id | default: publication.title | slugify -%} {%- if include.ids contains reference_id -%} {{- reference_separator -}}{{ publication_number }} {%- assign reference_separator = ", " -%} {%- endif -%} {%- else -%} {%- include publication-card.html publication=publication number=publication_number -%} {%- endif -%} {%- assign publication_number = publication_number | minus: 1 -%} {%- endfor -%} {%- endfor -%} {%- endfor -%} {%- endfor -%}
